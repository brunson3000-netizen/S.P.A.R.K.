//! TEMPORARY encoding-stability probe for the S2 evidence unification and
//! the m-02 manifest-hash correction. Not part of the canonical corpus:
//! it is run at the pre-change commit and at the post-change tree, and the
//! two outputs are compared. Removed from the tree after recording.
#![allow(
    clippy::unwrap_used,
    clippy::expect_used,
    clippy::panic,
    clippy::indexing_slicing,
    clippy::arithmetic_side_effects
)]

use spark_core::authority::Authority;
use spark_core::clock::LogicalTime;
use spark_core::hash::hash_bytes;
use spark_core::id::{CanonicalTag, CommandId, DefinitionId, ProfileId, SourceId};
use spark_core::scheduler::{
    DueWorkItem, OccurrenceIndex, Scheduler, WorkKey, WorkKind, WorkPayload,
    MAX_CONFLICT_TRACKED_CLAIMS,
};
use spark_core::scope::{ScopeId, ScopeKind};
use spark_core::timeline::{
    CommandKind, Ordinal, SemanticCommandEnvelope, TimelineEpoch, TimelineIngress,
    MAX_POISON_TRACKED_CLAIMS,
};
use spark_core::value::ValueConstraint;
use spark_engine::profile::definition::{DefinitionKind, DefinitionSpec};
use spark_engine::profile::manifest::ProfileManifest;
use spark_engine::profile::text::BoundedText;
use std::collections::BTreeSet;

fn profile() -> ProfileId {
    ProfileId::new("game-world").unwrap()
}
fn sequencer() -> SourceId {
    SourceId::new("sequencer.primary").unwrap()
}

fn item(index: usize) -> DueWorkItem {
    DueWorkItem {
        key: WorkKey {
            due_time: LogicalTime(7),
            profile_id: profile(),
            producer_definition_id: DefinitionId::new("rule.decay").unwrap(),
            scope_id: ScopeId::new(ScopeKind::Actor, "actor.a").unwrap(),
            occurrence_index: OccurrenceIndex(0),
            work_kind: WorkKind::new(CanonicalTag::new("work.decay").unwrap()),
        },
        payload: WorkPayload::new(hash_bytes(format!("payload.{index:05}").as_bytes())),
    }
}

fn envelope(index: usize) -> SemanticCommandEnvelope {
    let name = format!("cmd.{index:05}");
    SemanticCommandEnvelope {
        command_id: CommandId::new(&name).unwrap(),
        profile_id: profile(),
        timeline_epoch: TimelineEpoch(1),
        effective_time: LogicalTime(0),
        source_id: sequencer(),
        source_sequence: 0,
        input_ordinal: Ordinal(0),
        command_kind: CommandKind::new(CanonicalTag::new("test.command").unwrap()),
        canonical_payload_hash: hash_bytes(name.as_bytes()),
    }
}

fn scheduler_digest(count: usize) -> String {
    let mut scheduler = Scheduler::new();
    for index in 0..count {
        let _ = scheduler.schedule(item(index));
    }
    format!("{}", scheduler.canonical_state_digest())
}

fn timeline_digest(count: usize) -> String {
    let mut ingress = TimelineIngress::new(profile(), TimelineEpoch(1), sequencer(), 2).unwrap();
    for index in 0..count {
        let ticket = ingress.current_admission_window().unwrap().ticket();
        let _ = ingress.stage(&sequencer(), &envelope(index).submit_with(ticket));
    }
    format!("{}", ingress.canonical_state_digest())
}

fn definition(id: &str, description: &str) -> DefinitionSpec {
    DefinitionSpec {
        profile_id: profile(),
        id: DefinitionId::new(id).unwrap(),
        kind: DefinitionKind::Trait,
        domain: None,
        layer: None,
        value_constraint: ValueConstraint::boolean(),
        authority: Authority::SparkOwned,
        valid_scopes: BTreeSet::from([ScopeKind::Actor]),
        enabled: true,
        version: 1,
        description: BoundedText::new(description).unwrap(),
        behavioral_leverage: None,
    }
}

#[test]
fn print_canonical_digests() {
    for count in [2, 5, 17, 64, MAX_CONFLICT_TRACKED_CLAIMS, 300] {
        println!("scheduler_conflict[{count}] = {}", scheduler_digest(count));
    }
    for count in [2, 5, 17, 64, MAX_POISON_TRACKED_CLAIMS, 300] {
        println!("timeline_poison[{count}]    = {}", timeline_digest(count));
    }
    let curiosity = definition("trait.curiosity", "Curiosity trait.");
    let mut trust = definition("relationship.trust", "Trust relationship dimension.");
    trust.kind = DefinitionKind::Relationship;
    let one = ProfileManifest::new(
        profile(),
        BoundedText::new("1.0.0").unwrap(),
        vec![curiosity.clone()],
    );
    let two = ProfileManifest::new(
        profile(),
        BoundedText::new("1.0.0").unwrap(),
        vec![curiosity, trust],
    );
    println!("valid_manifest[1]         = {}", one.manifest_content_hash());
    println!("valid_manifest[2]         = {}", two.manifest_content_hash());
}
